cmake_minimum_required(VERSION 3.16)
project(actor_benchmarks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
add_compile_options(-fno-omit-frame-pointer)
enable_testing()
add_executable(benchmark_harness_tests harness_tests.cpp)
add_test(NAME benchmark_harness_tests COMMAND benchmark_harness_tests)
set_tests_properties(benchmark_harness_tests PROPERTIES TIMEOUT 5)
add_subdirectory(.. runtime EXCLUDE_FROM_ALL)
add_executable(native_echo native_echo.cpp)
target_link_libraries(native_echo PRIVATE actor_runtime)
add_executable(native_fiber_echo native_echo.cpp)
target_compile_definitions(native_fiber_echo PRIVATE BENCH_FIBER=1)
target_link_libraries(native_fiber_echo PRIVATE actor_runtime)
add_executable(wake_probe wake_probe.cpp)
target_link_libraries(wake_probe PRIVATE actor_runtime)
add_executable(native_ipc_echo ipc_echo.cpp)
target_link_libraries(native_ipc_echo PRIVATE actor_runtime)
add_executable(raw_ipc_echo ipc_echo.cpp)
target_compile_definitions(raw_ipc_echo PRIVATE BENCH_RAW_UNIX=1)
target_link_libraries(raw_ipc_echo PRIVATE actor_runtime)

set(BENCH_CAF_SOURCE "" CACHE PATH "Extracted CAF 0.19.6 sources")
set(BENCH_SO5_SOURCE "" CACHE PATH "Extracted SObjectizer v.5.8.3 sources")
if(BENCH_CAF_SOURCE)
  set(CAF_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CAF_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(CAF_ENABLE_IO_MODULE OFF CACHE BOOL "" FORCE)
  set(CAF_ENABLE_NET_MODULE OFF CACHE BOOL "" FORCE)
  set(CAF_ENABLE_OPENSSL_MODULE OFF CACHE BOOL "" FORCE)
  set(CAF_ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${BENCH_CAF_SOURCE} caf EXCLUDE_FROM_ALL)
  add_executable(caf_echo caf_echo.cpp)
  target_link_libraries(caf_echo PRIVATE CAF::core)
endif()
if(BENCH_SO5_SOURCE)
  set(SOBJECTIZER_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(SOBJECTIZER_BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(SOBJECTIZER_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory(${BENCH_SO5_SOURCE}/dev so5 EXCLUDE_FROM_ALL)
  add_executable(so5_echo so5_echo.cpp)
  target_link_libraries(so5_echo PRIVATE sobjectizer::StaticLib)
endif()
