cmake_minimum_required(VERSION 3.16)
project(actor_course_labs LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "请在 Linux/glibc 或 Ubuntu WSL 内构建课程实验")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -O0 -g)
add_executable(ch01_call_stack ch01_call_stack.cpp)
add_executable(ch02_state_machine ch02_state_machine.cpp)
add_executable(ch03_ucontext ch03_ucontext.cpp)
add_executable(ch02_build_it EXCLUDE_FROM_ALL exercises/ch02_build_it.cpp)
enable_testing()
add_test(NAME ch01_normal COMMAND ch01_call_stack)
add_test(NAME ch01_unwind COMMAND ch01_call_stack --throw)
add_test(NAME ch02_default COMMAND ch02_state_machine)
add_test(NAME ch02_variant COMMAND ch02_state_machine 10)
add_test(NAME ch03_suspend_resume COMMAND ch03_ucontext)
add_test(NAME ch03_exception_boundary COMMAND ch03_ucontext --throw)
set_tests_properties(ch01_normal ch01_unwind ch02_default ch02_variant
    ch03_suspend_resume ch03_exception_boundary PROPERTIES TIMEOUT 5)

find_package(Threads REQUIRED)
foreach(chapter IN ITEMS ch04_ready_queue ch05_waiting ch06_reclaim ch07_actor_token ch08_mailbox)
    add_executable(${chapter} ${chapter}.cpp)
    target_link_libraries(${chapter} PRIVATE Threads::Threads)
endforeach()
function(course_case name target mode)
    add_test(NAME ${name} COMMAND ${target} ${mode})
    set_tests_properties(${name} PROPERTIES TIMEOUT 10)
endfunction()
course_case(ch04_fifo ch04_ready_queue round-robin)
course_case(ch04_cooperative ch04_ready_queue no-yield)
course_case(ch04_empty ch04_ready_queue empty)
course_case(ch05_deadline ch05_waiting deadline)
course_case(ch05_tie ch05_waiting same-deadline)
course_case(ch05_zero ch05_waiting zero)
course_case(ch06_normal ch06_reclaim normal)
course_case(ch06_exception ch06_reclaim throw)
course_case(ch06_many ch06_reclaim many)
course_case(ch07_one_worker ch07_actor_token 1)
course_case(ch07_four_workers ch07_actor_token 4)
course_case(ch07_exception ch07_actor_token throw)
course_case(ch07_counterexample ch07_actor_token queue-only-model)
course_case(ch08_one_worker ch08_mailbox 1)
course_case(ch08_four_workers ch08_mailbox 4)
course_case(ch08_capacity ch08_mailbox capacity)
